Q1) Find the LCM of two numbers (a, b) and GCD of two numbers (b, c)
--------------------------------------------------------------------
File: LCMGCM.sh

Script:
echo "Enter number a: "
read a
echo "Enter number b: "
read b
echo "Enter number c: "
read c

# Calculate GCD of a and b
x=$a
y=$b
while [ $y -ne 0 ]; do
    temp=$y
    y=$((x % y))
    x=$temp
done
gcd_ab=$x

# Calculate LCM using GCD
lcm_ab=$(( (a * b) / gcd_ab ))

# Calculate GCD of b and c
x=$b
y=$c
while [ $y -ne 0 ]; do
    temp=$y
    y=$((x % y))
    x=$temp
done
gcd_bc=$x

# Display Results
echo "LCM of $a and $b is: $lcm_ab"
echo "GCD of $b and $c is: $gcd_bc"

Run:
chmod 777 LCMGCM.sh  
./LCMGCM.sh

Example Output:
Enter number a:  
6  
Enter number b:  
8  
Enter number c:  
12  
LCM of 6 and 8 is: 24  
GCD of 8 and 12 is: 4

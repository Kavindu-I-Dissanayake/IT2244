Q1) Multiplication Table using for loop
----------------------------------------
File: ForLoopMulTable.sh

Script:
echo 'Multiplication Table'
echo 'Enter number: '
read num

for ((i=1; i<=10; i++))
do
  a=$(($i * $num))
  echo -n "$i X $num : $a"
  echo ""
done

Run:
chmod 777 ForLoopMulTable.sh  
./ForLoopMulTable.sh

Q2) Print Diamond Pattern
--------------------------
File: Diomond_01.sh

Script:
echo "Enter number of rows: "
read rows

# Top half
for (( i=1; i<=rows; i++))
do
  for(( j=i; j<rows; j++))
  do
    echo -n " "
  done
  for (( k=1; k<=((2*i-1)); k++))
  do
    echo -n "*"
  done
  echo 
done	 

# Bottom half
for (( i=rows-1; i>=1; i--))
do
  for(( j=rows; j>i; j--))
  do
    echo -n " "
  done
  for (( k=1; k<=((2*i-1)); k++))
  do
    echo -n "*"
  done
  echo
done

Run:
chmod 777 Diomond_01.sh  
./Diomond_01.sh

Example Output:

    ***  
   *****  
  *******  
 *********  
  *******  
   *****  
    ***  
     *

Q3) Sum of the First 10 Fibonacci Numbers
-----------------------------------------
File: FibonacciSum.sh

Script:
a=0
b=1
sum=0

for (( i=0; i<10; i++))
do
  echo "$a"
  c=$((a + b))
  sum=$((sum + a))
  a=$b
  b=$c
done

echo "Sum of the first 10 Fibonacci numbers: $sum"

Run:
chmod 777 FibonacciSum.sh  
./FibonacciSum.sh

Example Output:

2  
3  
5  
8  
13  
21  
34  
Sum of the first 10 Fibonacci numbers: 88

Q4) Sum of Prime Numbers from 1 to 100
---------------------------------------
File: PrimeSum.sh

Script:
sum=0

for (( num=2; num<=100; num++ ))
do
  is_prime=1
  for (( i=2; i*i<=num; i++ ))
  do
    if (( num % i == 0 )); then
      is_prime=0
      break
    fi
  done

  if (( is_prime == 1 )); then
    sum=$((sum + num))
  fi
done

echo "Sum of prime numbers from 1 to 100 is: $sum"

Run:
chmod 777 PrimeSum.sh  
./PrimeSum.sh

Example Output:
Sum of prime numbers from 1 to 100 is: 1060
